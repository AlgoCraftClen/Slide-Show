    <script>
        const scripts = [
            "<strong>Intro:</strong> Welcome to the bid opening for 107,000 Jute Sacks. This is recorded for transparency.",
            "<strong>Committee:</strong> We have the GM, CPO, CFO, Ops, and HR present to witness the opening.",
            "<strong>Adachi:</strong> Decrypting Adachi's bid. Recording price now.",
            "<strong>Infinity:</strong> Decrypting Infinity's bid. Recording price now.",
            "<strong>AGRIX:</strong> Decrypting AGRIX's bid. Recording price now.",
            "<strong>Holland:</strong> Opening Holland's revised bid. Recording price now.",
            "<strong>Log:</strong> This is the preliminary price summary. No award is made today; technical evaluation follows."
        ];

        const slides = document.querySelectorAll('.slide');
        const scriptDisplay = document.getElementById('scriptDisplay');
        const currIdxDisplay = document.getElementById('currIdx');
        const prevBtn = document.getElementById('prevBtn');
        const nextBtn = document.getElementById('nextBtn');
        let idx = 0;

        function update() {
            slides.forEach((s, i) => s.classList.toggle('active', i === idx));
            scriptDisplay.innerHTML = scripts[idx];
            currIdxDisplay.innerText = idx + 1;
            prevBtn.disabled = idx === 0;
            nextBtn.disabled = idx === slides.length - 1;
        }

        prevBtn.addEventListener('click', () => { if(idx > 0) { idx--; update(); }});
        nextBtn.addEventListener('click', () => { if(idx < slides.length - 1) { idx++; update(); }});
        
        document.addEventListener('keydown', (e) => {
            if (e.target.tagName === 'INPUT') return;
            if (e.key === 'ArrowRight' || e.key === ' ') nextBtn.click();
            if (e.key === 'ArrowLeft') prevBtn.click();
        });

        update();
    </script>
